\section*{Phụ lục 1. Danh sách endpoint}

\begin{itemize}
    \item `GET /health`
    \item `GET /model`
    \item `POST /api/v1/vision/detect`
    \item `GET /api/v1/vision/detections/{detection_id}`
    \item `POST /api/v1/events/camera`
    \item `POST /analyze`
    \item `POST /analyze/mock`
\end{itemize}

\section*{Phụ lục 2. Gợi ý kiểm thử trình diễn}

\begin{enumerate}
    \item Mở dashboard tại `/`.
    \item Kiểm tra `/health`.
    \item Kiểm tra `/model`.
    \item Gửi một request thật tới `/api/v1/vision/detect`.
    \item Gửi một request mock nếu cần demo ổn định.
    \item Mở `openapi.yaml` hoặc Swagger UI để minh họa tài liệu API.
\end{enumerate}
